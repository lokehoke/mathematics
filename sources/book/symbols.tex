\chapter*{Справочник обозначений}
\addcontentsline{toc}{chapter}{Справочник обозначений}
Обозначения вводятся постепенно. Здесь можно быстро найти чтение символа; точные определения и условия применения приводятся в главах.
\renewcommand{\arraystretch}{1.3}
\small
\begin{longtable}{@{}p{.17\textwidth}p{.31\textwidth}p{.45\textwidth}@{}}
\rowcolor{Tint}\textbf{Символ}&\textbf{Как читать}&\textbf{Что означает и пример}\\\endfirsthead
\rowcolor{Tint}\textbf{Символ}&\textbf{Как читать}&\textbf{Что означает и пример}\\\endhead
\(=,\ne\)&равно; не равно&Равенство или различие: \(2+3=5\), \(0\ne1\).\\
\(<,>,\le,\ge\)&меньше; больше; не больше; не меньше&Сравнения: \(x\le3\) включает случай \(x=3\).\\
\(\approx\)&приблизительно равно&Округление: \(\sqrt2\approx1{,}414\).\\
\(+, -,\cdot,:\)&плюс; минус; умножить; разделить&\(6:3=2\); \(ab=a\cdot b\).\\
\(\frac ab\)&а разделить на б&Дробь, всегда \(b\ne0\).\\
\(\pm\)&плюс или минус&\(x=\pm2\) означает два значения \(-2\) и \(2\).\\
\(a^n,a^{-n}\)&а в степени эн&Степень; при отрицательном показателе \(a\ne0\).\\
\(\sqrt a,\sqrt[n]a\)&квадратный корень; корень энной степени&Для чётного \(n\) в действительных числах требуется \(a\ge0\).\\
\(|a|\)&модуль а&Расстояние до нуля: \(|-3|=3\).\\
\(a\mathrel{\vdots}b\)&а делится на б нацело&\(12\mathrel{\vdots}4\); делитель \(b\ne0\).\\
\(a\mid b\)&а делит б&\(4\mid12\). Порядок чисел противоположен записи с тремя точками.\\
\(a\nmid b\)&а не делит б&\(5\nmid12\).\\
\(\gcd,\operatorname{lcm}\)&джи-си-ди; эл-си-эм&НОД и НОК; например, \(\gcd(12,18)=6\).\\
\(\mathbb N,\mathbb N_0\)&натуральные; натуральные с нулём&В этой книге \(\mathbb N=\{1,2,\ldots\}\), \(\mathbb N_0=\{0,1,2,\ldots\}\).\\
\(\mathbb Z,\mathbb Q,\mathbb R\)&целые; рациональные; действительные&\(-2\in\mathbb Z\), \(\frac12\in\mathbb Q\), \(\sqrt2\in\mathbb R\).\\
\(\in,\notin\)&принадлежит; не принадлежит&\(2\in\{1,2,3\}\), \(4\notin\{1,2,3\}\).\\
\(\{\ldots\}\)&множество элементов&Фигурные скобки: \(A=\{1,2,3\}\).\\
\(\varnothing,U\)&пустое множество; универсум&\(\varnothing\) не содержит элементов; \(U\) задаёт рассматриваемые объекты.\\
\(A\subseteq B\)&а является подмножеством б&Каждый элемент \(A\) принадлежит \(B\); равенство разрешено.\\
\(A\subsetneq B\)&собственное подмножество&\(A\subseteq B\) и \(A\ne B\). Запись \(\subset\) в книгах бывает неоднозначной.\\
\(A\cup B,A\cap B\)&объединение; пересечение&Элементы хотя бы одного; элементы обоих множеств.\\
\(A\setminus B\)&разность множеств&Элементы \(A\), не принадлежащие \(B\).\\
\(A^c\)&дополнение а&\(U\setminus A\); зависит от выбранного универсума.\\
\(A\triangle B\)&симметрическая разность&Элементы ровно одного из \(A,B\).\\
\(A\times B\)&декартово произведение&Множество упорядоченных пар \((a,b)\), где \(a\in A,b\in B\).\\
\(\mathcal P(A),2^A\)&множество всех подмножеств&Для \(A=\{1\}\): \(\mathcal P(A)=\{\varnothing,\{1\}\}\).\\
\(|A|\)&число элементов а&Для конечного множества; отличается по контексту от модуля числа.\\
\((a;b),[a;b]\)&интервал; отрезок&Круглая скобка исключает конец, квадратная включает.\\
\(\infty\)&бесконечность&Не действительное число; в границах промежутков всегда круглая скобка.\\
\(\forall,\exists\)&для любого; существует&\(\forall x\in\mathbb R\;x^2\ge0\).\\
\(\Rightarrow,\Longleftrightarrow\)&следует; равносильно&Импликация; каждое из двух утверждений следует из другого.\\
\(\neg P\)&не пэ&Отрицание утверждения \(P\).\\
\(\land,\lor\)&и; или&Оба условия; хотя бы одно из условий.\\
\(S(n)\)&следующее за эн&Операция следования в аксиомах Пеано; \(S(0)=1\).\\
\(\sum_{k=1}^{n}a_k\)&сумма по ка от одного до эн&\(a_1+a_2+\ldots+a_n\).\\
\(\prod_{k=1}^{n}a_k\)&произведение по ка&\(a_1a_2\cdots a_n\).\\
\(n!\)&эн факториал&\(1\cdot2\cdots n\), отдельно \(0!=1\).\\
\(P_n,A_n^k\)&перестановки; размещения&Порядок выбранных объектов важен.\\
\(\binom nk,C_n^k\)&число сочетаний из эн по ка&Выбор \(k\) объектов из \(n\) без учёта порядка.\\
\(a_n,d,q,S_n\)&энный член; разность; знаменатель; сумма&Основные обозначения для последовательностей и прогрессий.\\
\(\lim_{n\to\infty}a_n=L\)&предел последовательности равен эль&Члены становятся сколь угодно близки к \(L\) при больших \(n\).\\
\(f:X\to Y\)&эф из икс в игрек&Отображение с областью определения \(X\) и множеством назначения \(Y\).\\
\(f(x),f\circ g\)&значение функции; композиция&\((f\circ g)(x)=f(g(x))\).\\
\(f^{-1}\)&обратная функция&Обратное отображение; обычно это не \(1/f\).\\
\(\exp x,e^x,\ln x\)&экспонента; е в степени икс; натуральный логарифм&\(\ln\) — логарифм по основанию \(e\).\\
\(\log_a x\)&логарифм икс по основанию а&\(a>0,a\ne1,x>0\).\\
\(\pi,\alpha\)&пи; альфа&\(\pi\) — отношение длины окружности к диаметру; \(\alpha\) часто обозначает угол.\\
\(\sin,\cos,\tan\)&синус; косинус; тангенс&В русской литературе тангенс также обозначают \(\operatorname{tg}\).\\
\(\arcsin\), \(\arccos\), \(\arctan\)&арксинус; арккосинус; арктангенс&Обратные функции на выбранных промежутках; \(\arctan\) также \(\operatorname{arctg}\).\\
\(\sinh,\cosh\)&гиперболический синус; гиперболический косинус&Другие записи: \(\operatorname{sh},\operatorname{ch}\).\\
\(\ast\)&звёздочка&Задача повышенной сложности.\\
\(\square\)&конец доказательства&После доказательства; в заданиях пустой квадрат может обозначать неизвестную цифру.\\
\end{longtable}
